    constructor() { this.init(); }

    init() {

      this.r  = base * (0.14 + Math.random() * 0.36);

    while (orbs.length > targetOrbs) orbs.pop();

    /* scale particle population to viewport */
    const targetP = Math.max(220, Math.min(760, Math.round((W * H) / 3100)));
    while (particles.length < targetP) particles.push(new Particle());
    while (particles.length > targetP) particles.pop();
  }

  /* ══════════════════════════════════════════════════════════
     Render — background
     ══════════════════════════════════════════════════════════ */
  function drawBackground(t, dt) {
    bctx.globalCompositeOperation = 'source-over';
    bctx.fillStyle = '#04050d';
    bctx.fillRect(0, 0, W, H);

    bctx.globalCompositeOperation = 'lighter';

    for (let i = 0; i < orbs.length; i++) {
      orbs[i].update(t, dt);
      orbs[i].draw(bctx, t);
    }

    /* soft light following the pointer */
    if (mouse.active) {
      const R = 280;
      const g = bctx.createRadialGradient(mouse.x, mouse.y, 0, mouse.x, mouse.y, R);
      g.addColorStop(0.0, 'rgba(170,205,255,0.13)');
      g.addColorStop(0.5, 'rgba(140,170,255,0.05)');
      g.addColorStop(1.0, 'rgba(140,170,255,0)');
      bctx.fillStyle = g;
      bctx.fillRect(mouse.x - R, mouse.y - R, R * 2, R * 2);
    }

    bctx.globalCompositeOperation = 'source-over';
  }

  /* ══════════════════════════════════════════════════════════
     Render — foreground
     ══════════════════════════════════════════════════════════ */
  function drawForeground(t, dt) {
    /* fade existing trails toward transparency */
    fctx.globalAlpha = 1;
    fctx.globalCompositeOperation = 'destination-out';
    fctx.fillStyle = 'rgba(0,0,0,0.055)';
    fctx.fillRect(0, 0, W, H);

    fctx.globalCompositeOperation = 'lighter';

    for (let i = 0; i < particles.length; i++) {
      const p = particles[i];
      p.update(t, dt);
      p.draw(fctx, t);
    }

    /* sparks */
    for (let i = sparks.length - 1; i >= 0; i--) {
      const s = sparks[i];
      s.x += s.vx * dt;
      s.y += s.vy * dt;
      s.vx *= Math.pow(0.94, dt);
      s.vy *= Math.pow(0.94, dt);
      s.life -= s.decay * dt;
      if (s.life <= 0) { sparks.splice(i, 1); continue; }

      const sz = s.size * 2.8;
      fctx.globalAlpha = s.life * 0.9;
      fctx.drawImage(sprites[s.si], s.x - sz, s.y - sz, sz * 2, sz * 2);
    }

    fctx.globalAlpha = 1;
    fctx.globalCompositeOperation = 'source-over';
  }

  /* ══════════════════════════════════════════════════════════
     Main loop
     ══════════════════════════════════════════════════════════ */
  let last = 0;

  function frame(now) {
    const dt = Math.min((now - last) / 16.6667, 2.5);
    last = now;

    /* eased pointer */
    const k = Math.min(1, 0.14 * dt);
    mouse.x += (mouse.tx - mouse.x) * k;
    mouse.y += (mouse.ty - mouse.y) * k;

    drawBackground(now, dt);
    drawForeground(now, dt);

    requestAnimationFrame(frame);
  }

  /* ══════════════════════════════════════════════════════════
     Events
     ══════════════════════════════════════════════════════════ */
  window.addEventListener('pointermove', function (e) {
    mouse.tx = e.clientX;
    mouse.ty = e.clientY;
    if (!mouse.active) {
      mouse.x = e.clientX;
      mouse.y = e.clientY;
      mouse.active = true;
    }
  }, { passive: true });

  window.addEventListener('pointerdown', function (e) {
    spawnSparks(e.clientX, e.clientY, 30);
  }, { passive: true });

  document.addEventListener('mouseleave', function () {
    mouse.active = false;
    mouse.tx = mouse.ty = -9999;
  });

  window.addEventListener('resize', resize);

  window.addEventListener('blur', function () {
    last = performance.now();
  });

  /* ══════════════════════════════════════════════════════════
     Go
     ══════════════════════════════════════════════════════════ */
  buildSprites();
  resize();

  last = performance.now();
  requestAnimationFrame(frame);

})();
</script>
</body>
</html>
